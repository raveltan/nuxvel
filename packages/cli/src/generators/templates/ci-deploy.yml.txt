name: Deploy {{ env }}

on:
  push:
    branches: [main]

concurrency:
  group: deploy-{{ env }}
  cancel-in-progress: false

env:
  NUXT_AUTH_SECRET: ci-only-auth-secret-not-for-production

jobs:
  test:
    runs-on: ubuntu-latest
    strategy:
      matrix:
        shard: [1, 2, 3]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: npm
      - run: npm ci
      - run: cp .env.example .env
      - run: npx nuxvel test:functional --shard=${{ matrix.shard }}/3

  check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: npm
      - run: npm ci
      - run: cp .env.example .env
      - run: npx nuxvel db:check
      - name: Add the SSH key of the deploy user
        env:
          NUXVEL_SSH_KEY: ${{ secrets.NUXVEL_SSH_KEY }}
        run: |
          test -s .nuxvel/known_hosts || { echo "::error::Commit .nuxvel/known_hosts, which nuxvel server:setup writes"; exit 1; }
          mkdir -p ~/.ssh
          printf '%s\n' "$NUXVEL_SSH_KEY" > ~/.ssh/id_ed25519
          chmod 600 ~/.ssh/id_ed25519
      - run: npx nuxvel route:list --diff-env={{ env }}
      - name: Run the tests of the live release against the new migrations
        run: |
          releases=$(npx nuxvel release:list {{ env }} --json)
          live=$(printf '%s' "$releases" | jq -r '.releases[] | select(.live) | .commit // empty')
          if [ -z "$live" ]; then
            echo "No live release with a commit, so there is no release to test against"
          elif git diff --quiet "$live" HEAD -- server/database/migrations; then
            echo "No migration changed since the live release"
          else
            npx nuxvel test:compat --against="$live"
          fi

  build:
    runs-on: {{ runner }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: npm
      - run: npm ci
      - run: npx nuxvel build --artifact --platform=linux/{{ arch }}
      - uses: actions/upload-artifact@v4
        with:
          name: release
          path: dist/

  deploy:
    needs: [test, check, build]
    runs-on: ubuntu-latest
    environment: {{ env }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: npm
      - run: npm ci
      - uses: actions/download-artifact@v4
        with:
          name: release
          path: dist/
      - name: Add the SSH key of the deploy user
        env:
          NUXVEL_SSH_KEY: ${{ secrets.NUXVEL_SSH_KEY }}
        run: |
          mkdir -p ~/.ssh
          printf '%s\n' "$NUXVEL_SSH_KEY" > ~/.ssh/id_ed25519
          chmod 600 ~/.ssh/id_ed25519
      - run: npx nuxvel deploy {{ env }} --artifact="$(ls dist/*.tar.gz)"
