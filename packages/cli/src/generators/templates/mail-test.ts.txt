import { describe, expect, it, renderMail } from "@nuxvel/nuxt/testing";

describe("{{name}} mail", () => {
  it("renders its template to HTML and text", async () => {
    const { subject, html, text } = await renderMail("{{name}}", { to: "ada@example.com" });

    expect(subject).toBe("{{title}}");
    expect(html).toContain(">{{title}}</h1>");
    expect(html).toContain("<!--[if mso");
    expect(text).toContain("Open the app https://example.com");
  });
});
